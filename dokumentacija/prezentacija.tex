\section{Korisnici sistema}

\begin{frame}
\frametitle{Akteri}
\begin{itemize}
\item Korisnik
\item Blagajnik
\item Menadžer
\item Sistem administrator
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Slučajevi upotrebe -- pregled}
\begin{enumerate}
\item \textbf{Online kupovina i rezervacija karte} -- Korisnik
\item \textbf{Kupovina na blagajni} -- Blagajnik, Korisnik
\item \textbf{Registracija} -- Korisnik (Gost)
\item \textbf{Autorizacija} -- Svi akteri
\item \textbf{Upravljanje podacima} -- Sistem Administrator
\item \textbf{Dodavanje novih zaposlenih} -- Sistem Administrator, Menadžer
\item \textbf{Izmena repertoara} -- Menadžer, Sistem Administrator
\end{enumerate}
\end{frame}

\begin{frame}
\frametitle{Online kupovina i rezervacija karte}
\textbf{Akter:} Korisnik 

\vspace{8pt}
\begin{itemize}
\item Korisnik pregleda repertoar i bira film
\item Sistem prikazuje termine i cene karata
\item Korisnik bira termin i sedište
\item Korisnik bira kupovinu ili rezervaciju
\item Sistem šalje potvrdu o uspešnosti
\end{itemize}

\vspace{5pt}
\textbf{Kupovina:} Provera bonus poena $\rightarrow$ Plaćanje $\rightarrow$ E-karta\\
\textbf{Rezervacija:} Ažuriranje podataka $\rightarrow$  Potvrda o rezervaciji
\end{frame}

\begin{frame}
\frametitle{Kupovina na blagajni}
\textbf{Akteri:} Blagajnik, Korisnik 

\vspace{8pt}
\begin{itemize}
\item Korisnik prilazi blagajniku i bira opciju:
  \begin{itemize}
  \item Kupovina karte
  \item Preuzimanje rezervisane karte
  \item Kupovina grickalica i pića
  \end{itemize}
\item Sistem prikazuje ukupnu cenu
\item Korisnik bira način plaćanja (keš ili kartica)
\item Sistem obrađuje plaćanje
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Registracija i Autorizacija}
\begin{columns}[t]
\column{0.5\textwidth}
\textbf{Registracija}\\
{\small Akter: Korisnik (Gost)}
\begin{itemize}
\item Korisnik bira registraciju
\item Unosi podatke
\item Sistem validira podatke
\item Sistem šalje verifikacioni email
\item Korisnik potvrđuje - nalog aktivan
\end{itemize}

\column{0.5\textwidth}
\textbf{Autorizacija}\\
{\small Akteri: Svi korisnici sistema}
\begin{itemize}
\item Akter bira prijavu
\item Unosi kredencijala
\item Sistem proverava podatke
\item Sistem utvrđuje ulogu aktera
\item Preusmeravanje na odgovarajuću stranicu
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}
\frametitle{Dodavanje novih zaposlenih}
\textbf{Akteri:} Sistem Administrator, Menadžer

\vspace{8pt}
\begin{itemize}
\item Menadžer šalje zahtev administratoru
\item Administrator unosi podatke u sistem 
\item Sistem validira podatke i proverava duplikate
\item Sistem proverava privilegije za dodelu uloge
\item Kreiranje naloga i slanje kredencijala novom zaposlenom
\item Slanje potvrde o uspešnosti menadžeru
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Upravljanje podacima}
\textbf{Akter:} Sistem Administrator

\vspace{8pt}
\begin{itemize}
\item Administrator pritiska dugme "Upravljaj podacima"
\item Sistem prikazuje polje za unos SQL naredbe
\item Administrator unosi naredbu i pritiska "Izvrši"
\item Sistem izvršava naredbu i šalje potvrdu
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Izmena repertoara}
\textbf{Akteri:} Menadžer, Sistem Administrator

\vspace{8pt}
\begin{itemize}
\item Sistem Administrator pritiska dugme "Izmeni dostupne filmove":
  \begin{itemize}
  \item Dodavanje filma -- unos podataka i čuvanje
  \item Uklanjanje filma -- biranje filma i brisanje 
  \end{itemize}
\item Menadžer pritiska "Izmeni repertoar"
\item Sistem prikazuje listu termina
\item Menadžer dodeljuje filmove terminima
\item Menadžer čuva izmene -- sistem potvrđuje uspešnost
\end{itemize}
\end{frame}
